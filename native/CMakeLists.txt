cmake_minimum_required(VERSION 3.21)
project(tinypdf_text VERSION 1.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TEXT REQUIRED IMPORTED_TARGET harfbuzz>=14.2.1 harfbuzz-subset harfbuzz-cairo icu-uc>=78.3 icu-i18n fribidi>=1.0.16 freetype2 cairo>=1.18 cairo-png librsvg-2.0>=2.52)
add_executable(tinypdf-text main.cpp)
target_link_libraries(tinypdf-text PRIVATE PkgConfig::TEXT)
if(MSVC)
  target_compile_options(tinypdf-text PRIVATE /W4)
else()
  target_compile_options(tinypdf-text PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS tinypdf-text RUNTIME DESTINATION bin)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/helper_fault.cpp")
  add_executable(tinypdf-test-helper EXCLUDE_FROM_ALL ../tests/helper_fault.cpp)
endif()

if(WIN32)
  target_link_libraries(tinypdf-text PRIVATE ws2_32)
  if(TARGET tinypdf-test-helper)
    target_link_libraries(tinypdf-test-helper PRIVATE ws2_32)
  endif()
endif()
